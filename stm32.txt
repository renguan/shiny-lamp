void L_runingled(void)    
{
    u8 i;
    u8 j;

    while(1)
    {
        for(i=1; i<=led_num; i++)
        {
            for(j=1; j<=led_num; j++)
            {
                L_ws2812_rgb(j, j == i ? WS_WHITE : WS_DARK);
                R_ws2812_rgb(j, WS_DARK);
            }

            L_ws2812_refresh(led_num);
            R_ws2812_refresh(led_num);
            delay_ms(100);
        }

        for(i=led_num; i>0; i--)
        {
            for(j=1; j<=led_num; j++)
            {
                L_ws2812_rgb(j, WS_DARK);
                R_ws2812_rgb(j, j == i ? WS_WHITE : WS_DARK);
            }

            L_ws2812_refresh(led_num);
            R_ws2812_refresh(led_num);
            delay_ms(100);
        }
    }
}


void R_runingled(void)
{
	u8 i,j;
	
	for(j=1;j<7;j++)    
	 { 
		 for(i=1;i<7;i++)    
		{
			if(i==j)
				R_ws2812_rgb(i, WS_WHITE);
			else
				R_ws2812_rgb(i, WS_DARK);
		}	
		 R_ws2812_refresh(led_num);  
		 delay_ms(100);
	 }
	
	for(j=6;j>=1;j--)    
	 { 
		 for(i=6;i>=1;i--)    
		{
			if(i==j)
				R_ws2812_rgb(i, WS_WHITE);
			else
				R_ws2812_rgb(i, WS_DARK);
		}	
		 R_ws2812_refresh(led_num);  
		 delay_ms(100);
	 }
